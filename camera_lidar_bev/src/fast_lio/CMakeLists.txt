cmake_minimum_required(VERSION 3.16)
project(fast_lio)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)

# Native Ouster input does not require a Livox SDK or custom message package.
option(FASTLIO_ENABLE_LIVOX "Compile the optional upstream Livox input path" OFF)
set(FASTLIO_THREADS 2 CACHE STRING "OpenMP correspondence-search threads (1..4)")
if(NOT FASTLIO_THREADS MATCHES "^[1-4]$")
  message(FATAL_ERROR "FASTLIO_THREADS must be between 1 and 4")
endif()

set(dependencies rclcpp geometry_msgs nav_msgs diagnostic_msgs sensor_msgs std_msgs std_srvs
  visualization_msgs tf2_ros pcl_conversions)
if(FASTLIO_ENABLE_LIVOX)
  find_package(livox_ros_driver2 REQUIRED)
  list(APPEND dependencies livox_ros_driver2)
endif()

rosidl_generate_interfaces(${PROJECT_NAME} "msg/Pose6D.msg")
ament_export_dependencies(rosidl_default_runtime)
add_executable(fastlio_mapping src/laserMapping.cpp src/preprocess.cpp
  include/ikd-Tree/ikd_Tree.cpp)
target_include_directories(fastlio_mapping PRIVATE include ${PCL_INCLUDE_DIRS})
target_compile_definitions(fastlio_mapping PRIVATE
  ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/" MP_PROC_NUM=${FASTLIO_THREADS})
if(FASTLIO_THREADS GREATER 1)
  target_compile_definitions(fastlio_mapping PRIVATE MP_EN)
endif()
if(FASTLIO_ENABLE_LIVOX)
  target_compile_definitions(fastlio_mapping PRIVATE FASTLIO_HAS_LIVOX)
endif()
target_link_libraries(fastlio_mapping Eigen3::Eigen ${PCL_LIBRARIES}
  OpenMP::OpenMP_CXX Threads::Threads)
rosidl_get_typesupport_target(typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(fastlio_mapping "${typesupport_target}")
ament_target_dependencies(fastlio_mapping ${dependencies})

install(TARGETS fastlio_mapping DESTINATION lib/${PROJECT_NAME})
install(FILES LICENSE UPSTREAM_COMMIT UPSTREAM.md PORT_NOTES.md DESTINATION share/${PROJECT_NAME})
ament_package()
