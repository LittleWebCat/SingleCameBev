cmake_minimum_required(VERSION 3.16)
project(bev_path_tracking LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(pure_pursuit src/pure_pursuit.cpp)
target_compile_features(pure_pursuit PUBLIC cxx_std_17)
target_include_directories(pure_pursuit PUBLIC include)
target_compile_options(pure_pursuit PRIVATE -Wall -Wextra -Wpedantic)
add_executable(path_tracker src/path_tracker.cpp)
target_link_libraries(path_tracker pure_pursuit)
ament_target_dependencies(path_tracker rclcpp nav_msgs std_msgs)
target_compile_options(path_tracker PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(pure_pursuit_tests test/pure_pursuit_tests.cpp)
  target_link_libraries(pure_pursuit_tests pure_pursuit)
  add_test(NAME pure_pursuit_tests COMMAND pure_pursuit_tests)
endif()
install(TARGETS path_tracker DESTINATION lib/bev_path_tracking)
ament_package()
