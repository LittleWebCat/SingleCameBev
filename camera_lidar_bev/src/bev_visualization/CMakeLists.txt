cmake_minimum_required(VERSION 3.16)
project(bev_visualization LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(vision_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)

add_library(rgb_projection STATIC src/rgb_projection.cpp src/cloud_io.cpp src/object_fusion.cpp src/camera_box_association.cpp)
target_compile_features(rgb_projection PUBLIC cxx_std_17)
target_include_directories(rgb_projection PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(rgb_projection PUBLIC ${OpenCV_LIBRARIES})
ament_target_dependencies(rgb_projection PUBLIC sensor_msgs)
target_compile_options(rgb_projection PRIVATE -Wall -Wextra -Wpedantic)

add_executable(camera_rgb_cloud src/camera_rgb_cloud.cpp)
target_compile_features(camera_rgb_cloud PRIVATE cxx_std_17)
target_compile_options(camera_rgb_cloud PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(camera_rgb_cloud PRIVATE rgb_projection yaml-cpp::yaml-cpp)
ament_target_dependencies(camera_rgb_cloud PUBLIC rclcpp sensor_msgs diagnostic_msgs tf2_msgs)
install(TARGETS camera_rgb_cloud DESTINATION lib/${PROJECT_NAME})

add_executable(camera_object_bev src/camera_object_bev.cpp)
target_compile_features(camera_object_bev PRIVATE cxx_std_17)
target_compile_options(camera_object_bev PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(camera_object_bev PRIVATE rgb_projection yaml-cpp::yaml-cpp)
ament_target_dependencies(camera_object_bev PUBLIC rclcpp sensor_msgs diagnostic_msgs tf2_msgs vision_msgs visualization_msgs geometry_msgs)
install(TARGETS camera_object_bev DESTINATION lib/${PROJECT_NAME})

add_executable(camera_lidar_box_fusion src/camera_lidar_box_fusion.cpp)
target_compile_features(camera_lidar_box_fusion PRIVATE cxx_std_17)
target_compile_options(camera_lidar_box_fusion PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(camera_lidar_box_fusion PRIVATE rgb_projection yaml-cpp::yaml-cpp)
ament_target_dependencies(camera_lidar_box_fusion PUBLIC rclcpp sensor_msgs diagnostic_msgs vision_msgs visualization_msgs geometry_msgs)
install(TARGETS camera_lidar_box_fusion DESTINATION lib/${PROJECT_NAME})

option(BEV_ENABLE_TENSORRT "Build the NVIDIA TensorRT camera object detector" ON)
if(BEV_ENABLE_TENSORRT)
  find_package(CUDAToolkit REQUIRED)
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATHS /usr/include/aarch64-linux-gnu /usr/include)
  find_library(TENSORRT_LIBRARY nvinfer PATHS /usr/lib/aarch64-linux-gnu)
  if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY)
    message(FATAL_ERROR "Camera object detection requires the JetPack-matched TensorRT headers/library. Set BEV_ENABLE_TENSORRT=OFF to build geometry/RGB without the GPU detector.")
  endif()
  add_library(yolo_detector STATIC src/yolo_detector.cpp)
  target_compile_features(yolo_detector PUBLIC cxx_std_17)
  target_include_directories(yolo_detector PUBLIC include ${OpenCV_INCLUDE_DIRS} ${TENSORRT_INCLUDE_DIR})
  target_compile_options(yolo_detector PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(yolo_detector PUBLIC ${OpenCV_LIBRARIES} ${TENSORRT_LIBRARY} CUDA::cudart)
  add_executable(camera_object_detector src/camera_object_detector.cpp)
  target_compile_features(camera_object_detector PRIVATE cxx_std_17)
  target_compile_options(camera_object_detector PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(camera_object_detector PRIVATE yolo_detector)
  ament_target_dependencies(camera_object_detector PUBLIC rclcpp sensor_msgs diagnostic_msgs vision_msgs)
  install(TARGETS camera_object_detector DESTINATION lib/${PROJECT_NAME})
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(rgb_projection_tests test/rgb_projection_tests.cpp)
  target_link_libraries(rgb_projection_tests PRIVATE rgb_projection)
  add_test(NAME rgb_projection_tests COMMAND rgb_projection_tests)
  add_executable(object_fusion_tests test/object_fusion_tests.cpp)
  target_link_libraries(object_fusion_tests PRIVATE rgb_projection)
  add_test(NAME object_fusion_tests COMMAND object_fusion_tests)
  add_executable(camera_box_association_tests test/camera_box_association_tests.cpp)
  target_link_libraries(camera_box_association_tests PRIVATE rgb_projection)
  add_test(NAME camera_box_association_tests COMMAND camera_box_association_tests)
  if(BEV_ENABLE_TENSORRT)
    add_executable(yolo_postprocess_tests test/yolo_postprocess_tests.cpp)
    target_link_libraries(yolo_postprocess_tests PRIVATE yolo_detector)
    add_test(NAME yolo_postprocess_tests COMMAND yolo_postprocess_tests)
  endif()
endif()
ament_package()
