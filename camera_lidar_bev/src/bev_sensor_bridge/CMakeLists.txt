cmake_minimum_required(VERSION 3.16)
project(bev_sensor_bridge LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core videoio)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)

add_library(camera_support STATIC src/camera_support.cpp)
target_compile_features(camera_support PUBLIC cxx_std_17)
target_include_directories(camera_support PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(camera_support PUBLIC yaml-cpp::yaml-cpp ${OpenCV_LIBRARIES})
ament_target_dependencies(camera_support PUBLIC sensor_msgs std_msgs)
target_compile_options(camera_support PRIVATE -Wall -Wextra -Wpedantic)

add_executable(camera_bridge src/camera_bridge.cpp)
target_compile_features(camera_bridge PRIVATE cxx_std_17)
target_compile_options(camera_bridge PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(camera_bridge PRIVATE camera_support)
ament_target_dependencies(camera_bridge PUBLIC rclcpp sensor_msgs std_msgs diagnostic_msgs)
add_executable(sensor_probe src/sensor_probe.cpp)
target_compile_features(sensor_probe PRIVATE cxx_std_17)
target_compile_options(sensor_probe PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(sensor_probe rclcpp sensor_msgs std_msgs nav_msgs diagnostic_msgs)
install(TARGETS sensor_probe camera_bridge DESTINATION lib/${PROJECT_NAME})

include(CTest)
if(BUILD_TESTING)
  add_executable(camera_support_tests test/camera_support_tests.cpp)
  target_link_libraries(camera_support_tests camera_support)
  add_test(NAME camera_support_tests COMMAND camera_support_tests)
endif()
ament_package()
